# The differential probes (tools/difftest), built with Retro68 for PowerPC
# against the Multiversal Interfaces generated from Executor's definitions
# (MV_OUT) by build-retro68.sh: <probe>PPC.bin each.
cmake_minimum_required(VERSION 3.16)
project(DiffTest C)
set(MV_OUT "" CACHE PATH "make-multiverse.rb -G CIncludes output")
include_directories(BEFORE "${MV_OUT}/CIncludes" "${CMAKE_CURRENT_SOURCE_DIR}")
link_directories(BEFORE "${MV_OUT}/libppc")
add_application(aevt aevt.c aevt.r CREATOR "DfAe")
target_link_libraries(aevt ObjectSupportLib)
add_application(appr appr.c appr.r CREATOR "DfAp")
target_link_libraries(appr AppearanceLib CollectionsLib DialogsLib WindowsLib CFragManager)
add_application(ctl ctl.c CREATOR "DfCt")
target_link_libraries(ctl ControlsLib DialogsLib AppearanceLib CFragManager)
add_application(menu menu.c CREATOR "DfMn")

add_application(sys sys.c sys.r CREATOR "DfSy")
target_link_libraries(sys CFragManager)
add_application(text text.c CREATOR "DfTx")
target_link_libraries(text CFragManager)
add_application(wind wind.c CREATOR "DfWn")
target_link_libraries(wind WindowsLib AppearanceLib CFragManager)
add_application(textutil textutil.c CREATOR "DfTu")
target_link_libraries(textutil CFragManager)
add_application(clut clut.c CREATOR "DfCl")
add_application(wide wide.c CREATOR "DfWd")
